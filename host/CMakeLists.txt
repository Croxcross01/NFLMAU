cmake_minimum_required(VERSION 3.20)
project(madden_arcade_pc C CXX)

# The recompiled code is generated on the user's machine from their own game dump (see setup.py),
# so nothing derived from the game ships with this project.
set(PPC_DIR "${CMAKE_SOURCE_DIR}/../build/ppc" CACHE PATH "Folder with XenonRecomp output (ppc_recomp.*.cpp)")
set(XENONRECOMP_DIR "${CMAKE_SOURCE_DIR}/../build/XenonRecomp" CACHE PATH "XenonRecomp checkout (for XenonUtils and simde)")

if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "The recompiled code needs Clang (on Windows: clang-cl from Visual Studio's 'C++ Clang tools' component).")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB PPC_SOURCES "${PPC_DIR}/ppc_recomp.*.cpp" "${PPC_DIR}/ppc_func_mapping.cpp")
if(NOT PPC_SOURCES)
    message(FATAL_ERROR "No recompiled sources in ${PPC_DIR}. Run setup.py with your game dump first.")
endif()

# generated code: built as a static library with light optimisation (it is huge)
add_library(madden_ppc STATIC ${PPC_SOURCES})
target_include_directories(madden_ppc PUBLIC "${PPC_DIR}" "${XENONRECOMP_DIR}/XenonUtils" "${XENONRECOMP_DIR}/thirdparty/simde")
if(MSVC)
    target_compile_options(madden_ppc PRIVATE /O1 -Wno-everything -msse4.1 -mavx2 /bigobj)
else()
    target_compile_options(madden_ppc PRIVATE -O1 -w -msse4.1 -mavx2 -fno-strict-aliasing)
endif()

# Vulkan: loaded at runtime through volk (MIT), so no Vulkan SDK is needed to build or run.
include(FetchContent)
FetchContent_Declare(vkheaders GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git GIT_TAG v1.3.296)
FetchContent_Declare(volk GIT_REPOSITORY https://github.com/zeux/volk.git GIT_TAG vulkan-sdk-1.3.296.0)
FetchContent_Declare(glslang GIT_REPOSITORY https://github.com/KhronosGroup/glslang.git GIT_TAG 15.0.0)
FetchContent_Populate(vkheaders)
FetchContent_Populate(volk)
# glslang compiles the translated Xenos shaders (GLSL) to SPIR-V at runtime
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(glslang)
set(VOLK_SRC "${volk_SOURCE_DIR}/volk.c")
set_source_files_properties(${VOLK_SRC} PROPERTIES INCLUDE_DIRECTORIES "${vkheaders_SOURCE_DIR}/include")
if(WIN32)
    set_source_files_properties(${VOLK_SRC} PROPERTIES COMPILE_DEFINITIONS VK_USE_PLATFORM_WIN32_KHR)
endif()

file(GLOB HOST_SOURCES "${CMAKE_SOURCE_DIR}/src/*.cpp")
add_executable(madden_host ${HOST_SOURCES} ${VOLK_SRC} "${XENONRECOMP_DIR}/thirdparty/tiny-AES-c/aes.c")
target_include_directories(madden_host PRIVATE "${volk_SOURCE_DIR}" "${vkheaders_SOURCE_DIR}/include")
target_link_libraries(madden_host PRIVATE glslang SPIRV glslang-default-resource-limits)

# Audio: XMA is decoded with FFmpeg's "xmaframes" decoder from Xenia's FFmpeg fork (LGPL-2.1). It is not stored in
# this repository: build it with tools/build_ffmpeg.sh (or point FFMPEG_DIR at an install with include/ and lib/).
# Without it the game still runs, with silent XMA audio.
set(FFMPEG_DIR "" CACHE PATH "Install prefix of the xenia-project/FFmpeg build (libavcodec + libavutil)")
if(FFMPEG_DIR)
    target_include_directories(madden_host PRIVATE "${FFMPEG_DIR}/include")
    target_link_directories(madden_host PRIVATE "${FFMPEG_DIR}/lib")
    target_link_libraries(madden_host PRIVATE avcodec avutil)
    if(WIN32)
        target_link_libraries(madden_host PRIVATE bcrypt)
    endif()
else()
    message(WARNING "FFMPEG_DIR not set: building without XMA audio decoding (sound effects will be silent)")
    target_compile_definitions(madden_host PRIVATE MADDEN_NO_FFMPEG)
endif()
if(WIN32)
    target_link_libraries(madden_host PRIVATE user32 gdi32 ole32 winmm)
else()
    find_package(SDL2 REQUIRED)
    target_link_libraries(madden_host PRIVATE SDL2::SDL2 ${CMAKE_DL_LIBS} m)
endif()
target_include_directories(madden_host PRIVATE "${XENONRECOMP_DIR}/thirdparty/tiny-AES-c")
target_link_libraries(madden_host PRIVATE madden_ppc)
target_include_directories(madden_host PRIVATE "${XENONRECOMP_DIR}/thirdparty/tiny-AES-c")
if(MSVC)
    target_compile_options(madden_host PRIVATE -msse4.1 -mavx2 -Wno-everything)
else()
    target_compile_options(madden_host PRIVATE -msse4.1 -mavx2 -w -fno-strict-aliasing)
    target_link_options(madden_host PRIVATE -rdynamic)
    find_package(Threads REQUIRED)
    target_link_libraries(madden_host PRIVATE Threads::Threads)
endif()
